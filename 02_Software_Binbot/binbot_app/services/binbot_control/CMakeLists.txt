cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(binbot_control_standalone C)
    set(CMAKE_C_STANDARD 11)
    include(CTest)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../libs/control_ipc" "control_ipc")
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_path(MOSQUITTO_INCLUDE_DIR mosquitto.h REQUIRED)
find_library(MOSQUITTO_LIBRARY NAMES mosquitto_static mosquitto REQUIRED)
add_executable(binbot_control src/main.c)
target_include_directories(binbot_control PRIVATE "${MOSQUITTO_INCLUDE_DIR}")
target_link_libraries(binbot_control PRIVATE binbot_control_ipc "${MOSQUITTO_LIBRARY}"
    OpenSSL::SSL OpenSSL::Crypto Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(binbot_control PRIVATE -Wall -Wextra -Wpedantic)
